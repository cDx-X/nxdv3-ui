pragma solidity ^0.8.20;

import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {SafeERC20} from "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
import "../interfaces/INXDv3Protocol.sol";

interface INXDBurnable_Local {
    function burn(uint256 amount) external;
}

interface INXDv3ERC20_Local {
    function governance() external view returns (address);
}

/**
 * @title NXDv3StakingVault (Optimism-Ready, In-Place Freezing Dual-Reward Vault with Epoch Tracking)
 * @notice NXD staking vault with ETH and XEN reward streaming in 7-day cycles.
 *         Includes an autonomous 7-day reward-inflow epoch tracking system dedicated to charts.
 */
contract NXDv3StakingVault {
    using SafeERC20 for IERC20;

    error InvalidAmount();
    error NoRequest();
    error Cooldown();
    error Underflow();
    error Unauthorized();
    error EmergencyWithdrawDisabled();
    error EmergencyWithdrawActive();
    error Locked();
    error IdenticalTokens();
    error ZeroAddress();
    error InvalidRecipient();

    event Deposit(address indexed user, uint256 amount);
    event Withdraw(address indexed user, uint256 amount);
    event EmergencyWithdraw(address indexed user, uint256 amount);
    event WithdrawRequested(address indexed user, uint256 amount, uint256 canWithdrawAfterTimestamp);
    event RewardsClaimed(address indexed user, uint256 amount);
    event XenRewardsClaimed(address indexed user, uint256 amount);
    event ETHTransferFailed(address indexed user, uint256 amount);
    event XenTransferFailed(address indexed user, uint256 amount);
    event EmergencyWithdrawEnabledUpdated(bool enabled);
    event RewardStreamStartedETH(uint256 rewardRate, uint256 periodFinish);
    event RewardStreamStartedXEN(uint256 rewardRate, uint256 periodFinish);
    event RewardStreamPausedETH(uint256 unallocatedRecovered);
    event RewardStreamPausedXEN(uint256 unallocatedRecovered);

    struct UserInfo {
        uint256 amount;        // Staked NXD balance
        uint256 rewardDebt;    // ETH reward debt
        uint256 accruedETH;    // Realized ETH ready to be claimed
        uint256 rewardDebtXen; // XEN reward debt
        uint256 accruedXEN;    // Realized XEN ready to be claimed
    }

    struct WithdrawalRequest {
        uint256 amount;
        uint256 canWithdrawAfterTimestamp;
    }

    // Structure dedicated to frontend reward-inflow charts
    struct RewardEpochData {
        uint256 epoch;
        uint256 startTime;
        uint256 ethAmount;
        uint256 xenAmount;
    }

    uint256 public constant PRECISION = 1e18;
    uint256 public constant WITHDRAWAL_COOLDOWN = 1 days;
    uint256 public constant REWARD_DURATION = 7 days;

    // --- Concurrency guard (Reentrancy Guard) ---
    uint256 private constant _NOT_ENTERED = 1;
    uint256 private constant _ENTERED = 2;
    uint256 private _status = _NOT_ENTERED;

    IERC20 public immutable nxd;
    IERC20 public immutable xen;
    INXDv3Protocol public immutable nxdProtocol;

    mapping(address => UserInfo) public userInfo;
    mapping(address => WithdrawalRequest) public withdrawalRequests;

    uint256 public totalStaked;

    // --- ETH stream metrics and state ---
    uint256 public accEthPerShare;
    uint256 public lastProcessedETHBalance;
    uint256 public unallocatedRewardsETH;
    uint256 public periodFinishETH;
    uint256 public rewardRateETH;
    uint256 public lastUpdateTimeETH;

    // --- XEN stream metrics and state ---
    uint256 public accXenPerShare;
    uint256 public lastProcessedXENBalance;
    uint256 public unallocatedRewardsXEN;
    uint256 public periodFinishXEN;
    uint256 public rewardRateXEN;
    uint256 public lastUpdateTimeXen;

    bool public emergencyWithdrawEnabled = false;
    uint256 public nxdPenaltyBurned;

    // Epoch statistics and analytics variables
    uint256 public contractStartTimestamp;
    uint256 public cumulativeRewardsSinceStart;
    uint256 public cumulativeXenRewardsSinceStart;

    // Epoch number (0, 1, 2...) => amount of rewards that flowed into the vault in that epoch
    mapping(uint256 => uint256) public ethInflowByEpoch;
    mapping(uint256 => uint256) public xenInflowByEpoch;

    modifier lock() {
        if (_status == _ENTERED) revert Locked();
        _status = _ENTERED;
        _;
        _status = _NOT_ENTERED;
    }

    function locked() external view returns (bool) {
        return _status == _ENTERED;
    }

    constructor(IERC20 _nxd, IERC20 _xen, INXDv3Protocol _nxdProtocol) payable {
        if (address(_nxd) == address(0) || address(_xen) == address(0) || address(_nxdProtocol) == address(0)) {
            revert ZeroAddress();
        }
        if (address(_nxd) == address(_xen)) {
            revert IdenticalTokens();
        }

        nxd = _nxd;
        xen = _xen;
        nxdProtocol = _nxdProtocol;

        // Align the epoch start to midnight 00:00:00 UTC
        contractStartTimestamp = (block.timestamp / 1 days) * 1 days;

        lastProcessedETHBalance = address(this).balance;
        if (lastProcessedETHBalance > 0) {
            unallocatedRewardsETH = lastProcessedETHBalance;
            cumulativeRewardsSinceStart = lastProcessedETHBalance;
            ethInflowByEpoch[0] = lastProcessedETHBalance;
        }

        uint256 initialXen = _xen.balanceOf(address(this));
        lastProcessedXENBalance = initialXen;
        if (initialXen > 0) {
            unallocatedRewardsXEN = initialXen;
            cumulativeXenRewardsSinceStart = initialXen;
            xenInflowByEpoch[0] = initialXen;
        }
    }

    // =============================================================
    //                     MAIN STREAM LOGIC
    // =============================================================

    function updateVault() public lock {
        _updateVault();
    }

    function _updateVault() internal {
        _syncEthStream();
        _syncXenStream();
    }

    function _syncEthIndexAndBalance() internal {
        uint256 _totalStaked = totalStaked;
        if (_totalStaked > 0) {
            uint256 _periodFinish = periodFinishETH;
            uint256 _rewardRate = rewardRateETH;
            uint256 applicableTime = block.timestamp < _periodFinish ? block.timestamp : _periodFinish;
            uint256 _lastUpdate = lastUpdateTimeETH;

            if (applicableTime > _lastUpdate && _rewardRate > 0) {
                uint256 timeDelta = applicableTime - _lastUpdate;
                uint256 rewardAmount = timeDelta * _rewardRate;

                uint256 shareIncrease = (rewardAmount * PRECISION) / _totalStaked;
                if (shareIncrease > 0) {
                    accEthPerShare += shareIncrease;
                    uint256 actuallyDistributed = (shareIncrease * _totalStaked) / PRECISION;
                    if (rewardAmount > actuallyDistributed) {
                        unallocatedRewardsETH += (rewardAmount - actuallyDistributed);
                    }
                    lastUpdateTimeETH = applicableTime;
                } else if (applicableTime == _periodFinish) {
                    unallocatedRewardsETH += rewardAmount;
                    lastUpdateTimeETH = _periodFinish;
                }
            }
        } else {
            lastUpdateTimeETH = block.timestamp;
        }

        uint256 currentBalance = address(this).balance;
        uint256 _lastProcessed = lastProcessedETHBalance;
        if (currentBalance > _lastProcessed) {
            uint256 incoming = currentBalance - _lastProcessed;
            lastProcessedETHBalance = currentBalance;
            cumulativeRewardsSinceStart += incoming;
            unallocatedRewardsETH += incoming;

            // Record into the current 7-day epoch
            uint256 epoch = (block.timestamp - contractStartTimestamp) / REWARD_DURATION;
            ethInflowByEpoch[epoch] += incoming;
        }
    }

    function _syncEthStream() internal {
        _syncEthIndexAndBalance();

        uint256 _periodFinish = periodFinishETH;
        bool streamExpired = block.timestamp >= _periodFinish;

        if (!emergencyWithdrawEnabled && totalStaked > 0) {
            if (streamExpired && unallocatedRewardsETH >= REWARD_DURATION) {
                _rollEthStream();
            } else if (streamExpired && rewardRateETH > 0) {
                rewardRateETH = 0;
            }
        } else if (streamExpired && rewardRateETH > 0) {
            rewardRateETH = 0;
        }
    }

    function _syncXenIndexAndBalance() internal {
        uint256 _totalStaked = totalStaked;
        if (_totalStaked > 0) {
            uint256 _periodFinish = periodFinishXEN;
            uint256 _rewardRate = rewardRateXEN;
            uint256 applicableTime = block.timestamp < _periodFinish ? block.timestamp : _periodFinish;
            uint256 _lastUpdate = lastUpdateTimeXen;

            if (applicableTime > _lastUpdate && _rewardRate > 0) {
                uint256 timeDelta = applicableTime - _lastUpdate;
                uint256 rewardAmount = timeDelta * _rewardRate;

                uint256 shareIncrease = (rewardAmount * PRECISION) / _totalStaked;
                if (shareIncrease > 0) {
                    accXenPerShare += shareIncrease;
                    uint256 actuallyDistributed = (shareIncrease * _totalStaked) / PRECISION;
                    if (rewardAmount > actuallyDistributed) {
                        unallocatedRewardsXEN += (rewardAmount - actuallyDistributed);
                    }
                    lastUpdateTimeXen = applicableTime;
                } else if (applicableTime == _periodFinish) {
                    unallocatedRewardsXEN += rewardAmount;
                    lastUpdateTimeXen = _periodFinish;
                }
            }
        } else {
            lastUpdateTimeXen = block.timestamp;
        }

        uint256 currentBalance = xen.balanceOf(address(this));
        uint256 _lastProcessed = lastProcessedXENBalance;
        if (currentBalance > _lastProcessed) {
            uint256 incoming = currentBalance - _lastProcessed;
            lastProcessedXENBalance = currentBalance;
            cumulativeXenRewardsSinceStart += incoming;
            unallocatedRewardsXEN += incoming;

            // Record into the current 7-day epoch
            uint256 epoch = (block.timestamp - contractStartTimestamp) / REWARD_DURATION;
            xenInflowByEpoch[epoch] += incoming;
        }
    }

    function _syncXenStream() internal {
        _syncXenIndexAndBalance();

        uint256 _periodFinish = periodFinishXEN;
        bool streamExpired = block.timestamp >= _periodFinish;

        if (!emergencyWithdrawEnabled && totalStaked > 0) {
            if (streamExpired && unallocatedRewardsXEN >= REWARD_DURATION) {
                _rollXenStream();
            } else if (streamExpired && rewardRateXEN > 0) {
                rewardRateXEN = 0;
            }
        } else if (streamExpired && rewardRateXEN > 0) {
            rewardRateXEN = 0;
        }
    }

    function notifyRewards() external lock {
        _checkAuthorized();
        _syncEthIndexAndBalance();
        _syncXenIndexAndBalance();
        if (!emergencyWithdrawEnabled && totalStaked > 0) {
            _rollEthStream();
            _rollXenStream();
        }
    }

    function _pauseEthStream() internal {
        uint256 _periodFinish = periodFinishETH;
        uint256 _rewardRate = rewardRateETH;

        if (block.timestamp < _periodFinish && _rewardRate > 0) {
            uint256 remainingTime = _periodFinish - lastUpdateTimeETH;
            uint256 leftover = remainingTime * _rewardRate;
            unallocatedRewardsETH += leftover;
            emit RewardStreamPausedETH(leftover);
        }
        rewardRateETH = 0;
        periodFinishETH = 0;
        lastUpdateTimeETH = block.timestamp;
    }

    function _pauseEthStreamIfEmpty() internal {
        if (totalStaked == 0) {
            _pauseEthStream();
        }
    }

    function _pauseXenStream() internal {
        uint256 _periodFinish = periodFinishXEN;
        uint256 _rewardRate = rewardRateXEN;

        if (block.timestamp < _periodFinish && _rewardRate > 0) {
            uint256 remainingTime = _periodFinish - lastUpdateTimeXen;
            uint256 leftover = remainingTime * _rewardRate;
            unallocatedRewardsXEN += leftover;
            emit RewardStreamPausedXEN(leftover);
        }
        rewardRateXEN = 0;
        periodFinishXEN = 0;
        lastUpdateTimeXen = block.timestamp;
    }

    function _pauseXenStreamIfEmpty() internal {
        if (totalStaked == 0) {
            _pauseXenStream();
        }
    }

    function forceRestartRewardStreamETH() external lock {
        _checkAuthorized();
        _syncEthIndexAndBalance();
        _rollEthStream();
    }

    function forceRestartRewardStreamXEN() external lock {
        _checkAuthorized();
        _syncXenIndexAndBalance();
        _rollXenStream();
    }

    function _rollEthStream() internal {
        if (totalStaked == 0 || emergencyWithdrawEnabled) return;

        uint256 _periodFinish = periodFinishETH;
        uint256 _rewardRate = rewardRateETH;
        bool streamExpired = block.timestamp >= _periodFinish;
        uint256 leftover = 0;
        if (!streamExpired && _rewardRate > 0) {
            uint256 remainingTime = _periodFinish - lastUpdateTimeETH;
            leftover = remainingTime * _rewardRate;
        }

        uint256 totalReward = unallocatedRewardsETH + leftover;
        uint256 newRate = totalReward / REWARD_DURATION;

        if (newRate > 0) {
            rewardRateETH = newRate;
            unallocatedRewardsETH = totalReward - (newRate * REWARD_DURATION);
            lastUpdateTimeETH = block.timestamp;
            uint256 newPeriodFinish = block.timestamp + REWARD_DURATION;
            periodFinishETH = newPeriodFinish;
            emit RewardStreamStartedETH(newRate, newPeriodFinish);
        } else {
            rewardRateETH = 0;
            periodFinishETH = 0;
            unallocatedRewardsETH = totalReward;
        }
    }

    function _rollXenStream() internal {
        if (totalStaked == 0 || emergencyWithdrawEnabled) return;

        uint256 _periodFinish = periodFinishXEN;
        uint256 _rewardRate = rewardRateXEN;
        bool streamExpired = block.timestamp >= _periodFinish;
        uint256 leftover = 0;
        if (!streamExpired && _rewardRate > 0) {
            uint256 remainingTime = _periodFinish - lastUpdateTimeXen;
            leftover = remainingTime * _rewardRate;
        }

        uint256 totalReward = unallocatedRewardsXEN + leftover;
        uint256 newRate = totalReward / REWARD_DURATION;

        if (newRate > 0) {
            rewardRateXEN = newRate;
            unallocatedRewardsXEN = totalReward - (newRate * REWARD_DURATION);
            lastUpdateTimeXen = block.timestamp;
            uint256 newPeriodFinish = block.timestamp + REWARD_DURATION;
            periodFinishXEN = newPeriodFinish;
            emit RewardStreamStartedXEN(newRate, newPeriodFinish);
        } else {
            rewardRateXEN = 0;
            periodFinishXEN = 0;
            unallocatedRewardsXEN = totalReward;
        }
    }

    // =============================================================
    //                 EPOCH SYSTEM AND CHART VIEWS
    // =============================================================

    /**
     * @notice Returns the number of the current 7-day epoch (0, 1, 2...).
     */
    function currentEpoch() public view returns (uint256) {
        return (block.timestamp - contractStartTimestamp) / REWARD_DURATION;
    }

    /**
     * @notice Returns the timestamp (00:00:00 UTC) of the start of the given epoch.
     */
    function getEpochStartTime(uint256 epochId) public view returns (uint256) {
        return contractStartTimestamp + (epochId * REWARD_DURATION);
    }

    /**
     * @notice Fetches the ETH and XEN inflow data array for a given epoch range.
     */
    function getEpochsInflowData(
        uint256 fromEpoch,
        uint256 toEpoch
    ) public view returns (RewardEpochData[] memory data) {
        if (fromEpoch > toEpoch) {
            return new RewardEpochData[](0);
        }

        uint256 current = currentEpoch();
        if (toEpoch > current) {
            toEpoch = current;
        }

        if (fromEpoch > toEpoch) {
            return new RewardEpochData[](0);
        }

        uint256 count = toEpoch - fromEpoch + 1;
        data = new RewardEpochData[](count);

        for (uint256 i = 0; i < count; i++) {
            uint256 ep = fromEpoch + i;
            data[i] = RewardEpochData({
                epoch: ep,
                startTime: contractStartTimestamp + (ep * REWARD_DURATION),
                ethAmount: ethInflowByEpoch[ep],
                xenAmount: xenInflowByEpoch[ep]
            });
        }
    }

    /**
     * @notice Fetches the ETH and XEN inflow data for all epochs so far.
     */
    function getAllEpochsInflowData() external view returns (RewardEpochData[] memory) {
        return getEpochsInflowData(0, currentEpoch());
    }

    // =============================================================
    //                            VIEWS
    // =============================================================

    function pendingETH(address _user) external view returns (uint256) {
        UserInfo storage user = userInfo[_user];
        uint256 simulatedAcc = accEthPerShare;
        uint256 _totalStaked = totalStaked;

        if (_totalStaked > 0) {
            uint256 _periodFinish = periodFinishETH;
            uint256 _rewardRate = rewardRateETH;
            uint256 applicableTime = block.timestamp < _periodFinish ? block.timestamp : _periodFinish;
            uint256 _lastUpdate = lastUpdateTimeETH;

            if (applicableTime > _lastUpdate && _rewardRate > 0) {
                uint256 timeDelta = applicableTime - _lastUpdate;
                uint256 rewardAmount = timeDelta * _rewardRate;
                simulatedAcc += (rewardAmount * PRECISION) / _totalStaked;
            }
        }

        uint256 accumulated = (user.amount * simulatedAcc) / PRECISION;
        if (accumulated >= user.rewardDebt) {
            return (accumulated - user.rewardDebt) + user.accruedETH;
        }
        return user.accruedETH;
    }

    function pendingXEN(address _user) external view returns (uint256) {
        UserInfo storage user = userInfo[_user];
        uint256 simulatedAcc = accXenPerShare;
        uint256 _totalStaked = totalStaked;

        if (_totalStaked > 0) {
            uint256 _periodFinish = periodFinishXEN;
            uint256 _rewardRate = rewardRateXEN;
            uint256 applicableTime = block.timestamp < _periodFinish ? block.timestamp : _periodFinish;
            uint256 _lastUpdate = lastUpdateTimeXen;

            if (applicableTime > _lastUpdate && _rewardRate > 0) {
                uint256 timeDelta = applicableTime - _lastUpdate;
                uint256 rewardAmount = timeDelta * _rewardRate;
                simulatedAcc += (rewardAmount * PRECISION) / _totalStaked;
            }
        }

        uint256 accumulated = (user.amount * simulatedAcc) / PRECISION;
        if (accumulated >= user.rewardDebtXen) {
            return (accumulated - user.rewardDebtXen) + user.accruedXEN;
        }
        return user.accruedXEN;
    }

    // =============================================================
    //                 USER OPERATIONS (CEI)
    // =============================================================

    function deposit(uint256 _amount) public lock {
        if (emergencyWithdrawEnabled) revert EmergencyWithdrawActive();
        if (_amount == 0) revert InvalidAmount();

        _updateVault();

        UserInfo storage user = userInfo[msg.sender];
        _realizeRewards(user);

        nxd.safeTransferFrom(msg.sender, address(this), _amount);
        user.amount += _amount;
        totalStaked += _amount;

        if (totalStaked == _amount) {
            _rollEthStream();
            _rollXenStream();
        }

        user.rewardDebt = (user.amount * accEthPerShare) / PRECISION;
        user.rewardDebtXen = (user.amount * accXenPerShare) / PRECISION;

        emit Deposit(msg.sender, _amount);
    }

    function withdraw(uint256 _amount, bool acceptsPenalty) public lock {
        if (emergencyWithdrawEnabled) revert EmergencyWithdrawActive();
        if (_amount == 0) revert InvalidAmount();
        _updateVault();

        UserInfo storage user = userInfo[msg.sender];
        if (user.amount < _amount) revert Underflow();
        _realizeRewards(user);

        user.amount -= _amount;
        totalStaked -= _amount;

        _pauseEthStreamIfEmpty();
        _pauseXenStreamIfEmpty();

        user.rewardDebt = (user.amount * accEthPerShare) / PRECISION;
        user.rewardDebtXen = (user.amount * accXenPerShare) / PRECISION;

        if (acceptsPenalty) {
            uint256 userGets = (_amount * 3) / 4;
            uint256 penalty = _amount - userGets;

            nxd.safeTransfer(msg.sender, userGets);
            INXDBurnable_Local(address(nxd)).burn(penalty);
            nxdPenaltyBurned += penalty;
            emit Withdraw(msg.sender, _amount);
        } else {
            WithdrawalRequest storage request = withdrawalRequests[msg.sender];
            if (request.amount > 0) revert Cooldown();

            uint256 timestamp = block.timestamp + WITHDRAWAL_COOLDOWN;
            withdrawalRequests[msg.sender] = WithdrawalRequest(_amount, timestamp);
            emit WithdrawRequested(msg.sender, _amount, timestamp);
        }
    }

    function withdrawCooldown() public lock {
        if (emergencyWithdrawEnabled) revert EmergencyWithdrawActive();

        WithdrawalRequest storage request = withdrawalRequests[msg.sender];

        if (request.amount == 0) revert NoRequest();
        if (block.timestamp < request.canWithdrawAfterTimestamp) revert Cooldown();

        uint256 amount = request.amount;
        request.amount = 0;
        request.canWithdrawAfterTimestamp = 0;

        nxd.safeTransfer(msg.sender, amount);
        emit Withdraw(msg.sender, amount);
    }

    // =============================================================
    //                      CLAIMING REWARDS
    // =============================================================

    function claimRewards() external lock {
        _claimRewardsFor(msg.sender, msg.sender);
    }

    function claimRewardsTo(address _recipient) external lock {
        if (_recipient == address(0)) revert InvalidRecipient();
        _claimRewardsFor(msg.sender, _recipient);
    }

    function claimETH() external lock {
        _claimETHFor(msg.sender, msg.sender);
    }

    function claimETHTo(address _recipient) external lock {
        if (_recipient == address(0)) revert InvalidRecipient();
        _claimETHFor(msg.sender, _recipient);
    }

    function claimXEN() external lock {
        _claimXENFor(msg.sender, msg.sender);
    }

    function claimXENTo(address _recipient) external lock {
        if (_recipient == address(0)) revert InvalidRecipient();
        _claimXENFor(msg.sender, _recipient);
    }

    function _claimRewardsFor(address _user, address _recipient) internal {
        _updateVault();
        UserInfo storage user = userInfo[_user];
        _realizeRewards(user);

        _payoutETH(_recipient, user);
        _payoutXEN(_recipient, user);
    }

    function _claimETHFor(address _user, address _recipient) internal {
        _updateVault();
        UserInfo storage user = userInfo[_user];
        _realizeRewards(user);

        _payoutETH(_recipient, user);
    }

    function _claimXENFor(address _user, address _recipient) internal {
        _updateVault();
        UserInfo storage user = userInfo[_user];
        _realizeRewards(user);

        _payoutXEN(_recipient, user);
    }

    function _payoutETH(address _to, UserInfo storage user) internal {
        uint256 ethAmount = user.accruedETH;
        if (ethAmount > 0) {
            uint256 currentBalance = address(this).balance;
            uint256 toTransfer = ethAmount > currentBalance ? currentBalance : ethAmount;

            if (toTransfer > 0) {
                user.accruedETH = ethAmount - toTransfer;
                bool sent = _safeETHTransfer(_to, toTransfer);
                if (sent) {
                    emit RewardsClaimed(_to, toTransfer);
                } else {
                    user.accruedETH = ethAmount;
                    emit ETHTransferFailed(_to, toTransfer);
                }
            }
        }
    }

    function _payoutXEN(address _to, UserInfo storage user) internal {
        uint256 xenAmount = user.accruedXEN;
        if (xenAmount > 0) {
            uint256 currentXenBalance = xen.balanceOf(address(this));
            uint256 toTransfer = xenAmount > currentXenBalance ? currentXenBalance : xenAmount;

            if (toTransfer > 0) {
                user.accruedXEN = xenAmount - toTransfer;
                lastProcessedXENBalance = currentXenBalance - toTransfer;

                bool sent = _safeTokenTransfer(address(xen), _to, toTransfer);
                if (sent) {
                    emit XenRewardsClaimed(_to, toTransfer);
                } else {
                    user.accruedXEN = xenAmount;
                    lastProcessedXENBalance = currentXenBalance;
                    emit XenTransferFailed(_to, toTransfer);
                }
            }
        }
    }

    function _realizeRewards(UserInfo storage user) internal {
        uint256 accumulatedETH = (user.amount * accEthPerShare) / PRECISION;
        if (accumulatedETH > user.rewardDebt) {
            user.accruedETH += (accumulatedETH - user.rewardDebt);
        }
        user.rewardDebt = accumulatedETH;

        uint256 accumulatedXEN = (user.amount * accXenPerShare) / PRECISION;
        if (accumulatedXEN > user.rewardDebtXen) {
            user.accruedXEN += (accumulatedXEN - user.rewardDebtXen);
        }
        user.rewardDebtXen = accumulatedXEN;
    }

    // =============================================================
    //                       EMERGENCY MODE
    // =============================================================

    function emergencyWithdraw() public lock {
        if (!emergencyWithdrawEnabled) revert EmergencyWithdrawDisabled();

        _updateVault();

        UserInfo storage user = userInfo[msg.sender];
        _realizeRewards(user);

        uint256 stakedAmount = user.amount;
        uint256 pendingWithdrawal = withdrawalRequests[msg.sender].amount;
        uint256 totalRefund = stakedAmount + pendingWithdrawal;

        if (totalRefund == 0 && user.accruedETH == 0 && user.accruedXEN == 0) {
            revert InvalidAmount();
        }

        user.amount = 0;
        user.rewardDebt = 0;
        user.rewardDebtXen = 0;
        delete withdrawalRequests[msg.sender];

        if (stakedAmount > 0) {
            totalStaked -= stakedAmount;
            _pauseEthStreamIfEmpty();
            _pauseXenStreamIfEmpty();
        }

        if (totalRefund > 0) {
            nxd.safeTransfer(msg.sender, totalRefund);
            emit EmergencyWithdraw(msg.sender, totalRefund);
        }

        _payoutETH(msg.sender, user);
        _payoutXEN(msg.sender, user);
    }

    /**
     * @notice Toggles the emergency mode (In-Place Freezing).
     *         - When enabled (true): the reward streams are paused. Unused
     *           reserves remain SAFELY ON THE CONTRACT BALANCE (no way to withdraw them externally).
     *         - When disabled (false): if there are stakers in the vault, the reward streams
     *           are automatically resumed from the preserved reserves.
     */
    function setEmergencyWithdrawEnabled(bool _enabled) external lock {
        _checkAuthorized();
        if (emergencyWithdrawEnabled == _enabled) return;

        _updateVault();
        emergencyWithdrawEnabled = _enabled;

        if (_enabled) {
            // Freeze the streams - unused capital returns to unallocatedRewards
            _pauseEthStream();
            _pauseXenStream();
        } else {
            // Automatically resume streaming after disabling emergency mode
            if (totalStaked > 0) {
                _rollEthStream();
                _rollXenStream();
            }
        }

        emit EmergencyWithdrawEnabledUpdated(_enabled);
    }

    function _checkAuthorized() internal view {
        if (msg.sender == address(nxdProtocol)) {
            return;
        }
        try INXDv3ERC20_Local(address(nxd)).governance() returns (address nxdGov) {
            if (nxdGov != address(0) && msg.sender == nxdGov) {
                return;
            }
        } catch {}

        revert Unauthorized();
    }

    function _safeETHTransfer(address _to, uint256 _amount) internal returns (bool) {
        if (address(this).balance < _amount) return false;
        lastProcessedETHBalance = address(this).balance - _amount;

        (bool sent, ) = payable(_to).call{value: _amount}("");
        if (!sent) {
            lastProcessedETHBalance = address(this).balance;
        }
        return sent;
    }

    function _safeTokenTransfer(address token, address to, uint256 value) internal returns (bool) {
        (bool success, bytes memory data) = token.call(
            abi.encodeWithSelector(IERC20.transfer.selector, to, value)
        );
        return success && (data.length == 0 || abi.decode(data, (bool)));
    }
