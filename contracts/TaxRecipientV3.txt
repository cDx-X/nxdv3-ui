pragma solidity ^0.8.20;

import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {SafeERC20} from "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
import "../interfaces/INXDv3Protocol.sol";
import "../interfaces/IUniswapV2Router02.sol";
import "../interfaces/IUniswapV2Pair.sol";
import "../interfaces/IUniswapV2Factory.sol";
import "../interfaces/IV2Oracle.sol";

/**
 * @title TaxRecipientV3 (All-in-One Autonomous Tax Engine & LP Balancer)
 * @notice Centralizes all the logic of asynchronous NXD tax processing:
 *         1. Swap 4/7 NXD -> DXN with Slippage and TWAP Oracle protection
 *         2. Staking exactly 25% of the DXN received from the swap in DBXen (1/7 of the total tax)
 *         3. Balanced LP addition (remaining 3/7 NXD + 3/7 DXN) to Uniswap V2
 *         4. Periodic sweeping and staking of leftover DXN
 * @dev L2 (Optimism) safe – oracle failures do not block NXD token transfers.
 */
contract TaxRecipientV3 {
    using SafeERC20 for IERC20;

    // --- Custom errors ---
    error Unauthorized();
    error ZeroAddress();
    error InvalidSlippage();
    error InvalidPair();
    error OracleNotConfigured();
    error Expired();

    // --- Immutable constants and addresses ---
    IERC20 public immutable nxd;
    address public immutable protocol;
    IERC20 public constant dxn = IERC20(0x90386DA49028ef1eF18F47f0a740cAAB50782576);
    IUniswapV2Router02 public constant UNISWAP_V2_ROUTER = IUniswapV2Router02(0x4A7b5Da61326A6379179b40d00F57E5bbDC962c2);
    address public constant UNISWAP_V2_FACTORY = 0x0c3c1c532F1e39EdF36BE9Fe0bE1410313E074Bf;

    // --- Storage packing ---
    uint16 public slippageBps = 500; // 5% by default (500 / 10000)
    uint32 public taxExecutionCount;
    bool private _isProcessingTax;

    IV2Oracle public v2Oracle;

    // Automatic leftover sweep interval
    uint32 public constant SWEEP_INTERVAL = 100;

    // --- Events ---
    event TaxProcessed(uint256 dxnStaked, uint256 nxdAddedLp, uint256 dxnAddedLp, uint256 liquidityMinted);
    event NXDSwappedForDXN(uint256 nxdSold, uint256 dxnReceived);
    event LeftoversSwept(uint256 nxdReturned, uint256 dxnStaked);
    event SlippageChanged(uint16 newSlippageBps);
    event V2OracleUpdated(address indexed newOracle);

    modifier onlyProtocol() {
        if (msg.sender != protocol) revert Unauthorized();
        _;
    }

    modifier onlyAuthorized() {
        if (msg.sender != address(nxd) && msg.sender != protocol) revert Unauthorized();
        _;
    }

    modifier lockTaxProcessing() {
        if (_isProcessingTax) revert Unauthorized();
        _isProcessingTax = true;
        _;
        _isProcessingTax = false;
    }

    constructor(address _protocol) {
        if (_protocol == address(0)) revert ZeroAddress();
        nxd = IERC20(msg.sender);
        protocol = _protocol;

        // Permanent transfer approvals for DXN
        dxn.forceApprove(address(UNISWAP_V2_ROUTER), type(uint256).max);
        dxn.forceApprove(_protocol, type(uint256).max);
    }

    // --- View functions ---

    function uniswapV2Pair() public view returns (address) {
        return IUniswapV2Factory(UNISWAP_V2_FACTORY).getPair(address(nxd), address(dxn));
    }

    // --- Administrative and configuration functions ---

    function setSlippageBps(uint16 _slippageBps) external onlyProtocol {
        if (_slippageBps > 10000) revert InvalidSlippage();
        slippageBps = _slippageBps;
        emit SlippageChanged(_slippageBps);
    }

    function setV2Oracle(address _oracle) external onlyProtocol {
        if (_oracle == address(0)) revert ZeroAddress();
        v2Oracle = IV2Oracle(_oracle);
        nxd.forceApprove(address(UNISWAP_V2_ROUTER), type(uint256).max);
        emit V2OracleUpdated(_oracle);
    }

    function forceSweep() external onlyProtocol {
        _sweepLeftovers();
    }

    // --- Asynchronous tax processing ---

    function processTax() external onlyAuthorized lockTaxProcessing {
        _processTaxInternal(block.timestamp);
    }

    function processTax(uint256 deadline) external onlyAuthorized lockTaxProcessing {
        if (block.timestamp > deadline) revert Expired();
        _processTaxInternal(deadline);
    }

    function _processTaxInternal(uint256 deadline) internal {
        IV2Oracle oracle = v2Oracle;
        if (address(oracle) == address(0)) return; // Silently exit if the oracle is not configured yet

        address pair = uniswapV2Pair();
        if (pair == address(0)) return; // Silently exit if the pair does not exist yet

        uint256 totalNXD = nxd.balanceOf(address(this));
        uint256 dxnReceived = 0;

        // =============================================================
        //  STEP 1: Sell 4/7 NXD for DXN (Swap via Uniswap V2)
        // =============================================================
        if (totalNXD > 0) {
            uint256 sellNXDAmount = (totalNXD * 4) / 7;
            if (sellNXDAmount > 0) {
                // Attempt a safe oracle update
                if (oracle.canUpdate()) {
                    try oracle.update() {} catch {}
                }

                // Safely query the oracle in a try/catch block
                uint256 consultAmount = 0;
                try oracle.consult(address(nxd), sellNXDAmount) returns (uint256 quote) {
                    consultAmount = quote;
                } catch {
                    // If the oracle is not ready (NotInitialized, SequencerDown, GracePeriod, etc.),
                    // we abort Step 1 without throwing an error. NXD safely waits in the contract.
                    consultAmount = 0;
                }

                uint256 amountOutMin = consultAmount > 0
                    ? (consultAmount * (10000 - slippageBps)) / 10000
                    : 0;

                // The swap is performed ONLY when we have a valid price from the oracle
                if (amountOutMin > 0) {
                    address[] memory path = new address[](2);
                    path[0] = address(nxd);
                    path[1] = address(dxn);

                    uint256 dxnBefore = dxn.balanceOf(address(this));
                    try UNISWAP_V2_ROUTER.swapExactTokensForTokensSupportingFeeOnTransferTokens(
                        sellNXDAmount,
                        amountOutMin,
                        path,
                        address(this),
                        deadline
                    ) {
                        dxnReceived = dxn.balanceOf(address(this)) - dxnBefore;
                        emit NXDSwappedForDXN(sellNXDAmount, dxnReceived);
                    } catch {
                        // If the swap failed (e.g. temporary slippage), do not fail the user's transfer
                        dxnReceived = 0;
                    }
                }
            }
        }

        // =============================================================
        //  STEP 2: Stake 25% of the DXN received from the swap in DBXen
        // =============================================================
        uint256 dxnToStake = 0;
        if (dxnReceived > 0) {
            dxnToStake = dxnReceived / 4;
            if (dxnToStake > 0) {
                INXDv3Protocol(protocol).depositNoMint(dxnToStake);
            }
        }

        // =============================================================
        //  STEP 3: Balanced LP addition for the remaining NXD and DXN
        // =============================================================
        _addOptimalLiquidity(pair, dxnToStake, deadline);

        // =============================================================
        //  STEP 4: Periodic sweep of leftover DXN
        // =============================================================
        unchecked {
            if (++taxExecutionCount >= SWEEP_INTERVAL) {
                taxExecutionCount = 0;
                _sweepLeftovers();
            }
        }
    }

    /**
     * @dev Isolated stack frame for adding LP on Uniswap V2.
     */
    function _addOptimalLiquidity(
        address pair,
        uint256 dxnToStake,
        uint256 deadline
    ) internal {
        uint256 ourNXDBalance = nxd.balanceOf(address(this));
        uint256 ourDXNBalance = dxn.balanceOf(address(this));

        if (ourNXDBalance == 0 || ourDXNBalance == 0 || pair == address(0)) {
            return;
        }

        (uint256 amountADesired, uint256 amountBDesired) = _getOptimalAmounts(
            pair,
            ourNXDBalance,
            ourDXNBalance
        );

        if (amountADesired > 0 && amountBDesired > 0) {
            try this.executeAddLiquidityExternal(amountADesired, amountBDesired, deadline) 
                returns (uint256 amountA, uint256 amountB, uint256 liquidity) 
            {
                emit TaxProcessed(dxnToStake, amountA, amountB, liquidity);
            } catch {
                // LP addition must not block NXD transfers in case of temporary volatility
            }
        }
    }

    /**
     * @notice External wrapper function for adding liquidity, enabling a safe try/catch
     */
    function executeAddLiquidityExternal(
        uint256 amountADesired,
        uint256 amountBDesired,
        uint256 deadline
    ) external returns (uint256 amountA, uint256 amountB, uint256 liquidity) {
        if (msg.sender != address(this)) revert Unauthorized();

        uint256 slippageFactor = 10000 - slippageBps;

        return UNISWAP_V2_ROUTER.addLiquidity(
            address(nxd),
            address(dxn),
            amountADesired,
            amountBDesired,
            (amountADesired * slippageFactor) / 10000,
            (amountBDesired * slippageFactor) / 10000,
            address(this),
            deadline
        );
    }

    function _getOptimalAmounts(
        address pair,
        uint256 ourNXDBalance,
        uint256 ourDXNBalance
    ) private view returns (uint256 amountADesired, uint256 amountBDesired) {
        (uint112 res0, uint112 res1, ) = IUniswapV2Pair(pair).getReserves();

        (uint256 resNXD, uint256 resDXN) = address(nxd) < address(dxn)
            ? (uint256(res0), uint256(res1))
            : (uint256(res1), uint256(res0));

        if (resNXD == 0 || resDXN == 0) {
            return (ourNXDBalance, ourDXNBalance);
        }

        uint256 dxnOptimal = (ourNXDBalance * resDXN) / resNXD;
        if (dxnOptimal <= ourDXNBalance) {
            return (ourNXDBalance, dxnOptimal);
        } else {
            return ((ourDXNBalance * resNXD) / resDXN, ourDXNBalance);
        }
    }

    function _sweepLeftovers() internal {
        uint256 leftoverDXN = dxn.balanceOf(address(this));

        if (leftoverDXN > 0) {
            INXDv3Protocol(protocol).depositNoMint(leftoverDXN);
        }

        emit LeftoversSwept(0, leftoverDXN);
    }
}
